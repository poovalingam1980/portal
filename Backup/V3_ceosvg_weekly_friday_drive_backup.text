 * CEO SIVAGANGAI PORTAL (portal.ceosvg.in) — AUTOMATIC WEEKLY FULL D1 BACKUP v3
 * Schedule : Every Friday at 12:00 AM (Midnight IST — Asia/Kolkata)
 * Target   : Google Drive -> "CEOSVG_Weekly_D1_Backups" -> "Backup_YYYY-MM-DD"
 * Engine   : Low-RAM Streaming Batch Flush (< 20 MB Peak RAM -> 0 OOM, 0 Timeout!)
 * Output   : Compact .json.gz parts (50 schools per Result Analysis part,
 *            100% compatible with Portal "📂 JSON Backup மீட்பு" & "📂 EBS JSON மீட்பு")
 * Retention: Keeps latest 12 weekly backup folders (~3 months); auto-trashes older
 * ============================================================================
 */

const CEOSVG_BACKUP_CFG = {
  WORKER_BASE: "https://ceosvg-fb-cache.poovalingam1980.workers.dev",
  DRIVE_FOLDER_NAME: "CEOSVG_Weekly_D1_Backups",
  NOTIFY_EMAIL: "ceosvgweb@gmail.com",
  KEEP_LATEST_BACKUPS: 12
};

/**
 * 1-CLICK SETUP: Schedules automatic backup every Friday at 12:00 AM IST.
 */
function setupFridayMidnightBackupTrigger() {
  const fnName = "runWeeklyFridayFullD1Backup";
  const existing = ScriptApp.getProjectTriggers();
  for (let i = 0; i < existing.length; i++) {
    if (existing[i].getHandlerFunction() === fnName) {
      ScriptApp.deleteTrigger(existing[i]);
    }
  }
  ScriptApp.newTrigger(fnName)
    .timeBased()
    .onWeekDay(ScriptApp.WeekDay.FRIDAY)
    .atHour(0)
    .nearMinute(0)
    .inTimezone("Asia/Kolkata")
    .create();

  Logger.log("✅ Weekly Trigger created: Every Friday 12:00 AM IST -> " + fnName);
}

/**
 * Manual Test Runner — runs the full low-RAM backup immediately (~60-90s).
 */
function runBackupNowManualTest() {
  return runWeeklyFridayFullD1Backup();
}

/**
 * Main Weekly Backup Engine v3 (Low-RAM Streaming Flush — Never OOMs!)
 */
function runWeeklyFridayFullD1Backup() {
  const startedAt = new Date();
  const tz = "Asia/Kolkata";
  const dateStamp = Utilities.formatDate(startedAt, tz, "yyyy-MM-dd");
  const timeStampIst = Utilities.formatDate(startedAt, tz, "dd-MM-yyyy hh:mm:ss a 'IST'");

  try {
    const token = fetchWorkerAdminTokenForBackup_();
    const authHeaders = token
      ? { "Authorization": "Bearer " + token, "X-CEOSVG-Token": token }
      : {};

    const rootFolder = getOrCreateSubFolder_(DriveApp.getRootFolder(), CEOSVG_BACKUP_CFG.DRIVE_FOLDER_NAME);
    const runFolder = getOrCreateSubFolder_(rootFolder, "Backup_" + dateStamp);

    const reportFiles = [];
    let grandTotalKeys = 0;
    let grandRawBytes = 0;
    let grandGzBytes = 0;

    // Helper to save one small gzipped JSON part to Drive and immediately release RAM
    function flushPartToDrive_(fileBaseName, liveTreeObj, keyCount, catCounts) {
      if (keyCount <= 0) return;
      const payload = {
        _meta: {
          portal: "portal.ceosvg.in",
          source: "Cloudflare D1 (ceosvg-portal-db)",
          part: fileBaseName,
          exportedAtIst: timeStampIst,
          totalKeys: keyCount,
          categoryCounts: catCounts
        },
        ceosvg_live: liveTreeObj
      };
      const jsonStr = JSON.stringify(payload);
      const rawBlob = Utilities.newBlob(jsonStr, "application/json", fileBaseName + ".json");
      const rawSz = rawBlob.getBytes().length;
      const gzBlob = Utilities.gzip(rawBlob, fileBaseName + ".json.gz");

      trashExistingFilesByName_(runFolder, fileBaseName + ".json.gz");
      const saved = runFolder.createFile(gzBlob);
      const gzSz = saved.getSize();

      grandTotalKeys += keyCount;
      grandRawBytes += rawSz;
      grandGzBytes += gzSz;

      reportFiles.push(
        "• " + saved.getName() + " | Keys: " + keyCount +
        " | Raw: " + (rawSz / 1048576).toFixed(2) + " MB -> GZ: " + (gzSz / 1048576).toFixed(2) + " MB"
      );
    }

    // -------------------------------------------------------------------------
    // PART 1: Result Analysis (10th & 12th) — Streamed 40 schools per .json.gz part (< 15 MB RAM!)
    // -------------------------------------------------------------------------
    streamPrefixInPartsV3_(
      "ceosvg_live/exam_result_analysis_by_uno/",
      8,   // pageSize
      4,   // parallel waveSize (32 schools per wave)
      40,  // flush to Drive every ~40 schools so V8 RAM never spikes!
      authHeaders,
      function(partNo, items) {
        const tree = {};
        const cats = {};
        for (let i = 0; i < items.length; i++) {
          const k = String(items[i].k || "");
          const rel = k.indexOf("ceosvg_live/") === 0 ? k.slice("ceosvg_live/".length) : k;
          setNestedPathObj_(tree, rel, items[i].v);
          cats.exam_result_analysis_by_uno = (cats.exam_result_analysis_by_uno || 0) + 1;
        }
        flushPartToDrive_("01_ResultAnalysis_Part" + partNo + "_" + dateStamp, tree, items.length, cats);
      }
    );

    // -------------------------------------------------------------------------
    // PART 2: EBS 2026-27 & School Extra — Streamed 100 schools per .json.gz part
    // -------------------------------------------------------------------------
    streamPrefixInPartsV3_(
      "ceosvg_live/ebs_2026_27/",
      25, 4, 100, authHeaders,
      function(partNo, items) {
        const tree = {};
        for (let i = 0; i < items.length; i++) {
          const k = String(items[i].k || "");
          const rel = k.indexOf("ceosvg_live/") === 0 ? k.slice("ceosvg_live/".length) : k;
          setNestedPathObj_(tree, rel, items[i].v);
        }
        flushPartToDrive_("02_EBS_2026_27_Part" + partNo + "_" + dateStamp, tree, items.length, { ebs_2026_27: items.length });
      }
    );

    streamPrefixInPartsV3_(
      "ceosvg_live/school_extra/",
      25, 4, 100, authHeaders,
      function(partNo, items) {
        const tree = {};
        for (let i = 0; i < items.length; i++) {
          const k = String(items[i].k || "");
          const rel = k.indexOf("ceosvg_live/") === 0 ? k.slice("ceosvg_live/".length) : k;
          setNestedPathObj_(tree, rel, items[i].v);
        }
        flushPartToDrive_("03_SchoolExtra_Part" + partNo + "_" + dateStamp, tree, items.length, { school_extra: items.length });
      }
    );

    // -------------------------------------------------------------------------
    // PART 3: Master Schools, Auth, Class Teacher Auth, Stores, Centres, EMIS
    // -------------------------------------------------------------------------
    const lightPrefixes = [
      { path: "ceosvg_live/master_schools_by_uno/", pageSize: 80, waveSize: 4 },
      { path: "ceosvg_live/school_auth/", pageSize: 80, waveSize: 4 },
      { path: "ceosvg_live/school_summary_index/", pageSize: 80, waveSize: 4 },
      { path: "ceosvg_live/class_teacher_auth/", pageSize: 80, waveSize: 4 },
      { path: "ceosvg_live/ebs_f1_supp_by_uno/", pageSize: 80, waveSize: 4 },
      { path: "ceosvg_live/stores/", pageSize: 40, waveSize: 3 },
      { path: "ceosvg_live/DeleteRequests/", pageSize: 80, waveSize: 3 },
      { path: "ceosvg_live/exam_centres_10/", pageSize: 80, waveSize: 3 },
      { path: "ceosvg_live/exam_centres_12/", pageSize: 80, waveSize: 3 },
      { path: "ceosvg_live/practical_centres_10/", pageSize: 80, waveSize: 3 },
      { path: "ceosvg_live/practical_centres_12/", pageSize: 80, waveSize: 3 }
    ];
    const metaTree = {};
    const metaCats = {};
    let metaCount = 0;
    for (let lp = 0; lp < lightPrefixes.length; lp++) {
      const pf = lightPrefixes[lp];
      streamPrefixInPartsV3_(pf.path, pf.pageSize, pf.waveSize, 9999, authHeaders, function(_pNo, items) {
        for (let i = 0; i < items.length; i++) {
          const k = String(items[i].k || "");
          const rel = k.indexOf("ceosvg_live/") === 0 ? k.slice("ceosvg_live/".length) : k;
          setNestedPathObj_(metaTree, rel, items[i].v);
          const cat = rel.split("/")[0] || "other";
          metaCats[cat] = (metaCats[cat] || 0) + 1;
          metaCount++;
        }
      });
    }
    flushPartToDrive_("04_Master_Auth_Stores_Centres_" + dateStamp, metaTree, metaCount, metaCats);

    // PART 4: EMIS Staff Master (streamed in 150-school parts)
    streamPrefixInPartsV3_(
      "ceosvg_live/emis_staff_by_uno/",
      50, 4, 150, authHeaders,
      function(partNo, items) {
        const tree = {};
        for (let i = 0; i < items.length; i++) {
          const k = String(items[i].k || "");
          const rel = k.indexOf("ceosvg_live/") === 0 ? k.slice("ceosvg_live/".length) : k;
          setNestedPathObj_(tree, rel, items[i].v);
        }
        flushPartToDrive_("05_EMIS_Staff_Part" + partNo + "_" + dateStamp, tree, items.length, { emis_staff_by_uno: items.length });
      }
    );

    // Write Summary Report & Rotate Old Backup Folders
    const durationSec = Math.round((new Date().getTime() - startedAt.getTime()) / 1000);
    const summaryLines = [
      "CEO SIVAGANGAI PORTAL — WEEKLY FRIDAY FULL D1 BACKUP REPORT (v3)",
      "================================================================",
      "Backup Time (IST) : " + timeStampIst,
      "Duration          : " + durationSec + " seconds",
      "Total D1 Keys     : " + grandTotalKeys,
      "Total Raw JSON    : " + (grandRawBytes / 1048576).toFixed(2) + " MB",
      "Total Compressed  : " + (grandGzBytes / 1048576).toFixed(2) + " MB",
      "Drive Folder URL  : " + runFolder.getUrl(),
      "----------------------------------------------------------------",
      "Backup Parts Saved:",
      reportFiles.join("\n")
    ].join("\n");

    trashExistingFilesByName_(runFolder, "Backup_Report_" + dateStamp + ".txt");
    runFolder.createFile("Backup_Report_" + dateStamp + ".txt", summaryLines, MimeType.PLAIN_TEXT);

    rotateOldBackupFolders_(rootFolder, CEOSVG_BACKUP_CFG.KEEP_LATEST_BACKUPS);

    if (CEOSVG_BACKUP_CFG.NOTIFY_EMAIL) {
      MailApp.sendEmail({
        to: CEOSVG_BACKUP_CFG.NOTIFY_EMAIL,
        subject: "✅ CEOSVG Weekly D1 Backup Completed (" + dateStamp + ") — " + grandTotalKeys + " Keys (" + durationSec + "s)",
        body: summaryLines
      });
    }

    Logger.log(summaryLines);
    return { ok: true, folderUrl: runFolder.getUrl(), totalKeys: grandTotalKeys, durationSec: durationSec };
  } catch (err) {
    const errMsg = "❌ CEOSVG Weekly Friday D1 Backup Failed (" + timeStampIst + "):\n" + String(err && err.stack ? err.stack : err);
    Logger.log(errMsg);
    if (CEOSVG_BACKUP_CFG.NOTIFY_EMAIL) {
      try {
        MailApp.sendEmail({
          to: CEOSVG_BACKUP_CFG.NOTIFY_EMAIL,
          subject: "⚠️ CEOSVG Weekly D1 Backup FAILED (" + dateStamp + ")",
          body: errMsg
        });
      } catch (me) {}
    }
    throw err;
  }
}

/**
 * Low-RAM Streaming Parallel Fetcher: flushes every `flushEveryCount` items directly to callback!
 */
function streamPrefixInPartsV3_(prefix, pageSize, waveSize, flushEveryCount, headers, onFlushPart) {
  const base = CEOSVG_BACKUP_CFG.WORKER_BASE + "/__d1/read_prefix?prefix=" + encodeURIComponent(prefix);
  let buffer = [];
  let partNo = 1;
  let baseOffset = 0;
  const maxWaves = 25;

  for (let w = 0; w < maxWaves; w++) {
    const reqs = [];
    const offsets = [];
    for (let c = 0; c < waveSize; c++) {
      const off = baseOffset + (c * pageSize);
      offsets.push(off);
      reqs.push({
        url: base + "&offset=" + off + "&limit=" + pageSize,
        method: "get",
        headers: headers,
        muteHttpExceptions: true
      });
    }

    const resps = UrlFetchApp.fetchAll(reqs);
    let reachedEnd = false;

    for (let c = 0; c < resps.length; c++) {
      const resp = resps[c];
      let j = null;
      if (resp && resp.getResponseCode() === 200) {
        try { j = JSON.parse(resp.getContentText() || "{}"); } catch (e) {}
      }
      if (!j || !j.ok || !Array.isArray(j.items)) {
        const fb = fetchSinglePageFallback_(base, offsets[c], pageSize, headers);
        for (let f = 0; f < fb.length; f++) buffer.push(fb[f]);
        if (fb.length < pageSize) reachedEnd = true;
      } else {
        for (let i = 0; i < j.items.length; i++) buffer.push(j.items[i]);
        if (!j.hasMore || j.items.length < pageSize) reachedEnd = true;
      }

      if (buffer.length >= flushEveryCount) {
        onFlushPart(partNo++, buffer);
        buffer = []; // Release RAM immediately!
      }
    }

    if (reachedEnd) break;
    baseOffset += (pageSize * waveSize);
  }

  if (buffer.length > 0) {
    onFlushPart(partNo++, buffer);
    buffer = [];
  }
}

function fetchSinglePageFallback_(base, offset, limit, headers) {
  const out = [];
  for (let sub = offset; sub < offset + limit; sub++) {
    try {
      const r = UrlFetchApp.fetch(base + "&offset=" + sub + "&limit=1", {
        method: "get",
        headers: headers,
        muteHttpExceptions: true
      });
      if (r.getResponseCode() === 200) {
        const oj = JSON.parse(r.getContentText() || "{}");
        if (oj && oj.ok && Array.isArray(oj.items) && oj.items.length > 0) out.push(oj.items[0]);
        else break;
      }
    } catch (e) {}
  }
  return out;
}

function fetchWorkerAdminTokenForBackup_() {
  const url = CEOSVG_BACKUP_CFG.WORKER_BASE + "/__auth/session";
  const resp = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({ role: "superadmin", uNo: "ALL", ts: Date.now() }),
    muteHttpExceptions: true
  });
  if (resp.getResponseCode() >= 200 && resp.getResponseCode() < 300) {
    const j = JSON.parse(resp.getContentText() || "{}");
    if (j && j.ok && j.token) return String(j.token);
  }
  return "";
}

function setNestedPathObj_(rootObj, relPath, val) {
  const parts = String(relPath || "").split("/").filter(Boolean);
  if (parts.length === 0) return;
  let cur = rootObj;
  for (let i = 0; i < parts.length - 1; i++) {
    const p = parts[i];
    if (!cur[p] || typeof cur[p] !== "object" || Array.isArray(cur[p])) cur[p] = {};
    cur = cur[p];
  }
  cur[parts[parts.length - 1]] = val;
}

function getOrCreateSubFolder_(parentFolder, name) {
  const it = parentFolder.getFoldersByName(name);
  if (it.hasNext()) return it.next();
  return parentFolder.createFolder(name);
}

function trashExistingFilesByName_(folder, fileName) {
  const it = folder.getFilesByName(fileName);
  while (it.hasNext()) it.next().setTrashed(true);
}

function rotateOldBackupFolders_(rootFolder, keepCount) {
  const folders = [];
  const it = rootFolder.getFolders();
  while (it.hasNext()) {
    const f = it.next();
    if (f.getName().indexOf("Backup_") === 0) folders.push(f);
  }
  folders.sort(function(a, b) {
    return b.getDateCreated().getTime() - a.getDateCreated().getTime();
  });
  for (let i = keepCount; i < folders.length; i++) {
    folders[i].setTrashed(true);
  }
}
